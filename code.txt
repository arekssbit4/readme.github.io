package com.example.restorenew

import android.content.Intent
import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken

class LoginActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main) // используем activity_main.xml

        val etLogin: EditText = findViewById(R.id.etLogin)
        val etPassword: EditText = findViewById(R.id.etPassword)
        val btnLogin: Button = findViewById(R.id.btnLogin)

        btnLogin.setOnClickListener {
            val login = etLogin.text.toString()
            val password = etPassword.text.toString()

            if (login.isEmpty() || password.isEmpty()) {
                showToast("Введите логин и пароль")
                return@setOnClickListener
            }

            val users = loadUsersFromJson()
            val foundUser = users.find { it.login == login && it.password == password }

            if (foundUser == null) {
                showToast("Неверный логин или пароль")
                return@setOnClickListener
            }
            when (foundUser.role) {
                "cook" -> {
                    startActivity(Intent(this, CookActivity::class.java))
                    finish()
                }
                "waiter" ->
                {
                    showToast("Роль официант пока в разработке")
                }
            }
        }
    }

    private fun loadUsersFromJson(): List<User> {
        val jsonString = assets.open("users.json").bufferedReader().use { it.readText() }
        val typeToken = object : TypeToken<List<User>>() {}.type
        return Gson().fromJson(jsonString, typeToken)
    }

    private fun showToast(message: String) {
        Toast.makeText(this, message, Toast.LENGTH_SHORT).show()
    }
}

package com.example.restorenew

import android.content.Intent
import android.os.Bundle
import android.widget.*
import androidx.appcompat.app.AppCompatActivity
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import java.io.File

class CookActivity : AppCompatActivity() {

    private lateinit var etDishName: EditText
    private lateinit var etDishDescription: EditText
    private lateinit var etDishPrice: EditText
    private lateinit var etDishCategory: EditText
    private lateinit var etDishCookTime: EditText

    private lateinit var btnAddDish: Button
    private lateinit var btnUpdateDish: Button
    private lateinit var btnDeleteDish: Button
    private lateinit var btnBackFromCook: Button

    private lateinit var linerLayoutMenu: LinearLayout

    private var selectedDishIndex = -1
    private val dishesFile = "dishes.json"
    private var dishList = mutableListOf<Dish>()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_cook)

        initViews()
        loadDishesFromFile()
        setupClickListeners()
        updateMenuView()
    }

    private fun initViews() {
        etDishName = findViewById(R.id.etDishName)
        etDishDescription = findViewById(R.id.etDishDescription)
        etDishPrice = findViewById(R.id.etDishPrice)
        etDishCategory = findViewById(R.id.etDishCategory)
        etDishCookTime = findViewById(R.id.etDishCookTime)

        btnAddDish = findViewById(R.id.btnAddDish)
        btnUpdateDish = findViewById(R.id.btnUpdateDish)
        btnDeleteDish = findViewById(R.id.btnDeleteDish)
        btnBackFromCook = findViewById(R.id.btnBackFromCook)

        linerLayoutMenu = findViewById(R.id.linerLayoutMenu)
    }

    private fun setupClickListeners() {
        btnAddDish.setOnClickListener { addDish() }
        btnUpdateDish.setOnClickListener { updateDish() }
        btnDeleteDish.setOnClickListener { deleteDish() }
        btnBackFromCook.setOnClickListener { startActivity(Intent(this, LoginActivity::class.java))
            finish() }
    }

    private fun loadDishesFromFile() {
        val file = File(filesDir, dishesFile)
        if (!file.exists()) {
            val jsonString = assets.open("dishes.json").bufferedReader().use { it.readText() }
            file.writeText(jsonString)
        }

        val jsonString = file.readText()
        val typeToken = object : TypeToken<List<Dish>>() {}.type
        dishList = Gson().fromJson(jsonString, typeToken)
    }

    private fun saveDishesToFile() {
        val jsonString = Gson().toJson(dishList)
        File(filesDir, dishesFile).writeText(jsonString)
    }

    private fun addDish() {
        val name = etDishName.text.toString()
        val description = etDishDescription.text.toString()
        val price = etDishPrice.text.toString().toDoubleOrNull()
        val category = etDishCategory.text.toString()
        val cookTime = etDishCookTime.text.toString().toIntOrNull()

        if (name.isEmpty() || price == null || cookTime == null) {
            showToast("Заполните все поля правильно")
            return
        }

        val newId = if (dishList.isNotEmpty()) dishList.maxOf { it.id } + 1 else 1
        val newDish = Dish(newId, name, description, price, category, cookTime)

        dishList.add(newDish)
        saveDishesToFile()
        updateMenuView()
        clearFields()
        showToast("Блюдо добавлено")
    }

    private fun updateDish() {
        if (selectedDishIndex == -1) {
            showToast("Выберите блюдо для редактирования")
            return
        }

        val name = etDishName.text.toString()
        val description = etDishDescription.text.toString()
        val price = etDishPrice.text.toString().toDoubleOrNull()
        val category = etDishCategory.text.toString()
        val cookTime = etDishCookTime.text.toString().toIntOrNull()

        if (name.isEmpty() || price == null || cookTime == null) {
            showToast("Заполните все поля правильно")
            return
        }

        dishList[selectedDishIndex] = dishList[selectedDishIndex].copy(
            name = name,
            description = description,
            price = price,
            category = category,
            cookTime = cookTime
        )

        saveDishesToFile()
        updateMenuView()
        clearFields()
        showToast("Блюдо обновлено")
    }

    private fun deleteDish() {
        if (selectedDishIndex == -1) {
            showToast("Выберите блюдо для удаления")
            return
        }

        dishList.removeAt(selectedDishIndex)
        saveDishesToFile()
        updateMenuView()
        clearFields()
        showToast("Блюдо удалено")
    }

    private fun updateMenuView() {
        linerLayoutMenu.removeAllViews()

        for ((index, dish) in dishList.withIndex()) {
            val dishView = layoutInflater.inflate(R.layout.item_dish, null)

            val nameText = dishView.findViewById<TextView>(R.id.tvDishName)
            val infoText = dishView.findViewById<TextView>(R.id.tvDishInfo)

            nameText.text = dish.name
            infoText.text = "${dish.description} | ${dish.price}₽ | ${dish.cookTime} мин"

            dishView.setOnClickListener {
                selectedDishIndex = index
                etDishName.setText(dish.name)
                etDishDescription.setText(dish.description)
                etDishPrice.setText(dish.price.toString())
                etDishCategory.setText(dish.category)
                etDishCookTime.setText(dish.cookTime.toString())
                showToast("Выбрано: ${dish.name}")
            }

            linerLayoutMenu.addView(dishView)
        }
    }

    private fun clearFields() {
        etDishName.text.clear()
        etDishDescription.text.clear()
        etDishPrice.text.clear()
        etDishCategory.text.clear()
        etDishCookTime.text.clear()
        selectedDishIndex = -1
    }

    private fun showToast(message: String) {
        Toast.makeText(this, message, Toast.LENGTH_SHORT).show()
    }
}
