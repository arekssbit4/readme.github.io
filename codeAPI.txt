        [HttpDelete("{id}")]

        {
            var user = _dataService.GetUserById(id);   ТРЕТИЙ

        {
            if (!System.IO.File.Exists("users.json"))

                return NotFound("Файл не найден");

            byte[] fileBytes = System.IO.File.ReadAllBytes("users.json");     ТРЕТИЙ

        }

using System.Text.Json;
using Ekzgotovo.Models;

namespace Ekzgotovo.Services
{
    public class JsonDataService
    {
        private static readonly string JsonFilePath = "users.json";      ПЕРВЫЙ
        private static List<User> _users = new();

        public JsonDataService()
        {
            LoadFromJsonFile();
        }

        public bool ValidateUser(string login, string password)
        {
            return _users.Any(u => u.Login == login && u.Password == password);
        }

        public List<User> GetAllUsers() => _users;

        public User GetUserById(int id) => _users.FirstOrDefault(u => u.Id == id); 

        public void AddUser(User user)
        {
            int newId = _users.Count > 0 ? _users.Max(u => u.Id) + 1 : 1;
            user.Id = newId;
            _users.Add(user);
            SaveToJsonFile();                               ПЕРВЫЙ
        }

        public void UpdateUser(int id, User updatedUser)
        {
            var user = _users.FirstOrDefault(u => u.Id == id);
            if (user != null)
            {
                user.Login = updatedUser.Login;
                user.Password = updatedUser.Password;          ВТОРОЙ
                SaveToJsonFile();
            }
        }

        public void DeleteUser(int id)
        {
            var user = _users.FirstOrDefault(u => u.Id == id);
            if (user != null)
            {
                _users.Remove(user);
                SaveToJsonFile();                     ВТОРОЙ
            }
        }

        private void LoadFromJsonFile()
        {
            if (File.Exists(JsonFilePath))
            {
                string json = File.ReadAllText(JsonFilePath); 
                _users = JsonSerializer.Deserialize<List<User>>(json) ?? new List<User>();              ТРЕТИЙ   
            }
            else
            {
                _users = new List<User>
                {
                    new User { Id = 1, Login = "admin", Password = "123" }
                };
                SaveToJsonFile();
            }
        }

        private void SaveToJsonFile()
        {
            string json = JsonSerializer.Serialize(_users, new JsonSerializerOptions              ТРЕТИЙ
            {
                WriteIndented = true
            });
            File.WriteAllText(JsonFilePath, json);
        }
    }
}
